```fsharp
let rec compileExp (e : TypedExp)
				   (vtable : SymTab.SymTab<reg>)
				   (place : reg) : Instruction list =
match e with
// ... existing match cases ...
| IndexOf (e1, e2, tp, pos) -> 
	let v = newReg "v" // value being searched for
	let a = newReg "a" // the array
	let len = newReg "len" // length of array
	let elt = newReg "elt" // next array element
	let lloop = newLab "loop"
	let ldone = newLab "done"
	let code1 = compileExp e1 vtable v // searching for exp v
	let code2 = compileExp e2 vtable a // searching in arr a'
	let sz = getEltSize tp  // element size as ESByte/word
	let szi = sizeToInt sz  // element size as byte count
	let searchCode =
		[ LW (len a 0)  // fetch array length
		; ADDI (a, a, 4) // point to 1st element
		; LI (place 0) // init search index ('i' in the for loop)
		; LABEL lloop // start the search loop
		; BGE (place, len, ldone) // if i >= len, then done
		; Load sz (elt, a, 0) // fetch next element
		; BEQ (elt, v, ldone) // if v == next element, then done
		; ADDI (a, a, szi) // increment array pointer
		; ADDI (place, place, 1) // increment index in for loop
		; J lloop // restart loop
		; LABEL ldone ] // label done
	code1 @ code2 @ searchcode
```

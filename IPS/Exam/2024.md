mup : ( (α ∗ β → γ) ∗ [α] ∗ β ) → [γ].

```haskell
tf = lookup(ftable, getname(f)) -- (a * b -> y)
ta = CheckExp(expa, vtable, ftable) --[a]
tb = CheckExp(expb, vtable, ftable) -- b
te = case tb of -- error: should be a
	Array(t) => t
	| other => error("array expected: expa")
case tf of
	unbound => error("function undefined")
	| ((t1 * t2) -> t3) =>
		if (t1 = t3) then -- error: t3 should be [a]
			if (t2 = tb) then Tuple(t3) -- error: should be t3 not in a tuple
			else error("type mismatch: 2nd fct arg and expb")
		else error("type mismatch: 1st fct arg and expa")
	| other => error("binary function expected")
```
